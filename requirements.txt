# PostBridge — Python dependencies
# Install with:  pip install -r requirements.txt

# ── Core GUI (included with Python on Windows/macOS, may need separate install on Linux)
# tkinter is part of the stdlib; install python3-tk via your system package manager if missing

# ── Drag-and-drop support (optional but recommended)
tkinterdnd2>=0.3.0

# ── Whisper transcription backend
faster-whisper>=1.0.0

# ── AAF parsing (required for Script→AAF and AAF→XML workflows)
pyaaf2>=1.7.0

# ── Audio/video processing (ffmpeg must also be installed separately on PATH)
# ffmpeg-python is NOT required — PostBridge calls ffmpeg directly via subprocess

# ── NumPy for waveform blob detection (optional — gracefully disabled if absent)
numpy>=1.24.0

# ── Audio playback
# Live, callback-driven playback for the Step 4 waveform editor: edits and
# speed changes are heard without restarting.  Also the first playback
# backend that is not Windows-only.  If it is absent PostBridge falls back
# to rendering a WAV and handing it to winsound (Windows only, no live
# response), so this is strongly recommended rather than strictly required.
sounddevice>=0.4.6
